set(PRISM_HARDENING -Wall -Wextra -Wpedantic -fstack-protector-strong)
function(prism_harden target)
 target_compile_options(${target} PRIVATE ${PRISM_HARDENING} ${ARGN})
 target_compile_definitions(${target} PRIVATE $<$<NOT:$<CONFIG:Debug>>:_FORTIFY_SOURCE=3>)
 set_property(TARGET ${target} PROPERTY POSITION_INDEPENDENT_CODE ON)
 get_target_property(kind ${target} TYPE)
 if(NOT kind STREQUAL "STATIC_LIBRARY")
  target_link_options(${target} PRIVATE -Wl,-z,relro,-z,now,-z,noexecstack)
 endif()
endfunction()
