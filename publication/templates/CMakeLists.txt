cmake_minimum_required(VERSION 3.22)
project(KyPrism VERSION 0.13.0 LANGUAGES CXX)
include(GNUInstallDirs)
include(CTest)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
# GTK 4 window; the decoder/cache/file engine uses only non-visual Qt modules.
# Qt Widgets is needed solely by the optional legacy front end and its tests.
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Gui Concurrent)
find_package(Qt6 6.4 QUIET COMPONENTS Widgets OpenGLWidgets Test)
find_package(PkgConfig REQUIRED)
find_package(ZLIB REQUIRED)
pkg_check_modules(VIPS REQUIRED IMPORTED_TARGET vips-cpp>=8.12)
pkg_check_modules(RAW REQUIRED IMPORTED_TARGET libraw>=0.20)
pkg_check_modules(GTK4 REQUIRED IMPORTED_TARGET gtk4>=4.12)
pkg_check_modules(EPOXY REQUIRED IMPORTED_TARGET epoxy)
option(PRISM_BUILD_QT_LEGACY "Build the previous Qt Widgets front end (ky-prism-qt) and its tests" ON)
if(PRISM_BUILD_QT_LEGACY AND NOT (Qt6Widgets_FOUND AND Qt6OpenGLWidgets_FOUND AND Qt6Test_FOUND))
  message(STATUS "Qt Widgets/OpenGLWidgets/Test not found: legacy Qt front end and its tests are skipped")
  set(PRISM_BUILD_QT_LEGACY OFF)
endif()
add_library(prism_engine STATIC
  src/raster_source.cpp src/language.cpp src/document.cpp src/prefetch.cpp
  src/thumbnail_cache.cpp src/thumbnail_disk.cpp src/nvjpeg_backend.cpp
  src/pixel_simd.cpp src/favorites.cpp src/file_ops_core.cpp)
target_include_directories(prism_engine PUBLIC src)
target_link_libraries(prism_engine PUBLIC Qt6::Core Qt6::Gui Qt6::Concurrent PkgConfig::VIPS PkgConfig::RAW ZLIB::ZLIB ${CMAKE_DL_LIBS})
add_library(prism_gtk STATIC
  src/gtk/window.cpp src/gtk/photo_view.cpp src/gtk/library_pane.cpp
  src/gtk/dialogs.cpp src/gtk/table.cpp src/gtk/theme.cpp src/gtk/icons.cpp
  src/gtk/ui_text.cpp src/gtk/settings.cpp)
set_target_properties(prism_gtk PROPERTIES AUTOMOC OFF)
target_include_directories(prism_gtk PUBLIC src src/gtk)
target_link_libraries(prism_gtk PUBLIC prism_engine PkgConfig::GTK4 PkgConfig::EPOXY)
target_compile_definitions(prism_gtk PUBLIC PRISM_VERSION="${PROJECT_VERSION}")
add_executable(ky-prism src/gtk/main.cpp)
set_target_properties(ky-prism PROPERTIES AUTOMOC OFF)
target_link_libraries(ky-prism PRIVATE prism_gtk)
foreach(t prism_engine prism_gtk ky-prism)
  target_compile_options(${t} PRIVATE -Wall -Wextra -Wpedantic -fstack-protector-strong)
  target_compile_definitions(${t} PRIVATE $<$<NOT:$<CONFIG:Debug>>:_FORTIFY_SOURCE=2>)
  set_property(TARGET ${t} PROPERTY POSITION_INDEPENDENT_CODE ON)
endforeach()
# GTK may deprecate API between minor releases; that must not break a build.
target_compile_options(prism_gtk PRIVATE -Wno-deprecated-declarations)
target_compile_options(ky-prism PRIVATE -Wno-deprecated-declarations)
target_link_options(ky-prism PRIVATE -Wl,-z,relro,-z,now,-z,noexecstack)

set(PRISM_WITH_NVJPEG AUTO CACHE STRING "AUTO: use local CUDA SDK if available; ON: require it; OFF: CPU JPEG decoder")
set_property(CACHE PRISM_WITH_NVJPEG PROPERTY STRINGS AUTO ON OFF)
if(NOT PRISM_WITH_NVJPEG MATCHES "^(AUTO|ON|OFF)$")
  message(FATAL_ERROR "PRISM_WITH_NVJPEG must be AUTO, ON or OFF")
endif()
if(NOT PRISM_WITH_NVJPEG STREQUAL OFF)
  find_package(CUDAToolkit 12 QUIET)
  if(CUDAToolkit_FOUND AND TARGET CUDA::nvjpeg)
    include(CheckLanguage)
    check_language(CUDA)
  endif()
endif()
if(NOT PRISM_WITH_NVJPEG STREQUAL OFF AND CUDAToolkit_FOUND AND TARGET CUDA::nvjpeg AND CMAKE_CUDA_COMPILER)
  if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
    set(CMAKE_CUDA_ARCHITECTURES "75-real;89-real;75-virtual")
  endif()
  enable_language(CUDA)
  add_library(prism_nvjpeg MODULE src/nvjpeg_cuda.cpp src/nvjpeg_resize.cu)
  set_target_properties(prism_nvjpeg PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON POSITION_INDEPENDENT_CODE ON INSTALL_RPATH_USE_LINK_PATH TRUE)
  target_link_libraries(prism_nvjpeg PRIVATE CUDA::nvjpeg CUDA::cudart)
  file(WRITE "${CMAKE_BINARY_DIR}/nvjpeg-build.json" "{\"enabled\":true,\"toolkit\":\"${CUDAToolkit_VERSION}\"}\n")
  install(TARGETS prism_nvjpeg LIBRARY DESTINATION libexec/ky-prism)
  message(STATUS "nvJPEG available: local CUDA ${CUDAToolkit_VERSION}; targets ${CMAKE_CUDA_ARCHITECTURES}; runtime AUTO with CPU fallback")
elseif(PRISM_WITH_NVJPEG STREQUAL ON)
  message(FATAL_ERROR "nvJPEG requires your separately installed NVIDIA CUDA Toolkit >=12 with nvcc and nvJPEG. Use AUTO to allow CPU fallback.")
else()
  file(WRITE "${CMAKE_BINARY_DIR}/nvjpeg-build.json" "{\"enabled\":false}\n")
  message(STATUS "nvJPEG SDK unavailable or disabled: libvips CPU decoding; OpenGL rendering remains enabled")
endif()

install(TARGETS ky-prism RUNTIME DESTINATION libexec/ky-prism)
install(PROGRAMS ky-prism DESTINATION bin)
install(FILES "${CMAKE_BINARY_DIR}/nvjpeg-build.json" DESTINATION libexec/ky-prism)
install(FILES ky-prism.desktop DESTINATION share/applications)
install(FILES ky-prism.svg DESTINATION share/icons/hicolor/scalable/apps)
install(FILES LICENSE COPYRIGHT.md THIRD_PARTY_NOTICES.md DESTINATION share/licenses/ky-prism)
install(FILES src/NVIDIA-NVJPEG-NOTICE.txt DESTINATION share/licenses/ky-prism)
install(DIRECTORY LICENSES/ DESTINATION share/licenses/ky-prism/third-party)

# Also available in release installs: verifies actual decoding, not just dlopen.
add_executable(prism-nvjpeg-test tests/nvjpeg.cpp)
target_link_libraries(prism-nvjpeg-test PRIVATE prism_engine)
install(TARGETS prism-nvjpeg-test RUNTIME DESTINATION libexec/ky-prism)

if(BUILD_TESTING)
  find_package(OpenGL REQUIRED COMPONENTS OpenGL EGL)
  # GTK front end. The window test needs a display; without one it skips (77).
  add_executable(prism-gtk-settings-test tests/gtk_settings.cpp src/gtk/settings.cpp)
  set_target_properties(prism-gtk-settings-test PROPERTIES AUTOMOC OFF)
  target_include_directories(prism-gtk-settings-test PRIVATE src src/gtk)
  target_link_libraries(prism-gtk-settings-test PRIVATE Qt6::Core)
  add_test(NAME gtk_settings COMMAND prism-gtk-settings-test)

  target_compile_options(prism-gtk-test PRIVATE -Wno-deprecated-declarations)
  add_test(NAME gtk_ui COMMAND prism-gtk-test)
  # Label "display": a real window on the session's compositor. The installer
  # excludes it (ctest -LE display); run `ctest -L display` to exercise it.

    ENVIRONMENT "PRISM_JPEG_BACKEND=cpu;PRISM_LANGUAGE=pl;PRISM_UI_OPACITY=")
  # Engine tests: no widget toolkit.
  function(prism_engine_test name)

    target_include_directories(prism-${name}-test PRIVATE src)

    add_test(NAME ${name} COMMAND prism-${name}-test)
    set_tests_properties(${name} PROPERTIES ENVIRONMENT "PRISM_JPEG_BACKEND=cpu;PRISM_LANGUAGE=pl" TIMEOUT 60)
  endfunction()
  prism_engine_test(decode)
  prism_engine_test(raster_source)
  add_executable(prism-pixel_simd-test tests/pixel_simd.cpp src/pixel_simd.cpp)
  target_include_directories(prism-pixel_simd-test PRIVATE src)
  add_test(NAME pixel_simd COMMAND prism-pixel_simd-test)
  foreach(name shader thumbnail_shader)
    add_executable(prism-${name}-test tests/${name}.cpp)
    target_include_directories(prism-${name}-test PRIVATE src)
    target_link_libraries(prism-${name}-test PRIVATE OpenGL::OpenGL OpenGL::EGL)
    add_test(NAME ${name} COMMAND prism-${name}-test)
    set_tests_properties(${name} PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 30)
  endforeach()
  add_test(NAME nvjpeg_cpu COMMAND prism-nvjpeg-test --cpu)
  add_test(NAME nvjpeg_gpu COMMAND prism-nvjpeg-test)
  set_tests_properties(nvjpeg_gpu PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 60)
  add_executable(prism-auto-fallback-test tests/auto_fallback.cpp)
  target_link_libraries(prism-auto-fallback-test PRIVATE prism_engine)
  set_target_properties(prism-auto-fallback-test PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/no-adapter/bin")
  add_test(NAME auto_cpu_fallback COMMAND prism-auto-fallback-test)
  set_tests_properties(auto_cpu_fallback PROPERTIES TIMEOUT 30)

  # Previous Qt Widgets front end: built as ky-prism-qt (not installed) so its
  # event tests keep guarding the shared engine.
  if(PRISM_BUILD_QT_LEGACY)
    set(PRISM_QT_ENV "QT_QPA_PLATFORM=offscreen;QT_QPA_PLATFORMTHEME=;QT_STYLE_OVERRIDE=Fusion;PRISM_JPEG_BACKEND=cpu;PRISM_LANGUAGE=pl")
    add_library(prism_raster INTERFACE)
    target_link_libraries(prism_raster INTERFACE prism_engine Qt6::Widgets)
    add_library(prism_decode INTERFACE)
    target_link_libraries(prism_decode INTERFACE prism_engine Qt6::Widgets)
    add_library(prism_favorites STATIC src/thumbnails.cpp src/thumbnail_gpu.cpp src/favorites_ui.cpp src/file_ops.cpp src/ui_extensions.cpp src/photo_dialog.cpp src/usb_access.h)

    set_property(TARGET prism_favorites PROPERTY POSITION_INDEPENDENT_CODE ON)
    add_executable(ky-prism-qt src/main.cpp src/view.cpp)

    function(prism_test name)
      add_executable(prism-${name}-test tests/${name}.cpp ${ARGN})
      target_include_directories(prism-${name}-test PRIVATE src)
